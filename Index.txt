        function processDashboardData(headersRaw, rows) {
            const headers = headersRaw.map(h => String(h).trim().toLowerCase());
            let leaderIdx = headers.findIndex(h => h === 'leader'); let memberIdx = headers.findIndex(h => h === 'member');
            let crmIdx = headers.findIndex(h => h === 'crm order type'); let errIdx = headers.findIndex(h => h === 'errorcode');
            let subErrIdx = headers.findIndex(h => h === 'suberrorcode'); let memoIdx = headers.findIndex(h => h.includes('engineermemo'));
            let jenisIdx = headers.findIndex(h => h === 'jenis_error' || h === 'status');
            let addrIdx = headers.findIndex(h => h === 'address');
            
            // Kolom untuk Modal:
            let woIdx = headers.findIndex(h => h.includes('workorder'));
            let tselIdx = headers.findIndex(h => h === 'order id tsel' || h.includes('sc order no') || h.includes('csrm'));
            let custIdx = headers.findIndex(h => h === 'customer name');

            // Reset Filtered Data Modal
            window.dashFilteredData = { total: {all:[], create:[], migrate:[], modify:[]}, kp: {all:[], create:[], migrate:[], modify:[]}, kt: {all:[], create:[], migrate:[], modify:[]}, op: {all:[], create:[], migrate:[], modify:[]}, comp: {all:[], create:[], migrate:[], modify:[]} };

            let stats = { total: {all:0, create:0, migrate:0, modify:0}, kp: {all:0, create:0, migrate:0, modify:0}, kt: {all:0, create:0, migrate:0, modify:0}, op: {all:0, create:0, migrate:0, modify:0}, comp: {all:0, create:0, migrate:0, modify:0}, timSummary: {} };
            let mapPoints = [];

            // Helper format Leader name for Grouping = TIM_1 (Leader - Member)
            let timMap = {};
            mappingTimCache.forEach(t => timMap[String(t.leader).trim()] = `${String(t.tim).trim()} (${String(t.leader).trim()} - ${String(t.member).trim()})`);

            rows.forEach((row, idx) => {
                let leader = leaderIdx !== -1 ? String(row[leaderIdx] || "").trim() : "";
                let member = memberIdx !== -1 ? String(row[memberIdx] || "").trim() : "";
                
                // ORDER HI RULE: Must be assigned
                if (!leader || leader === "-" || !member || member === "-") return;

                let crmRaw = crmIdx !== -1 ? String(row[crmIdx] || "").trim().toLowerCase() : "";
                let type = "other";
                if (crmRaw.includes("create")) type = "create";
                else if (crmRaw.includes("migrate")) type = "migrate";
                else if (crmRaw.includes("modify")) type = "modify";

                let err = errIdx !== -1 ? String(row[errIdx] || "") : ""; let sub = subErrIdx !== -1 ? String(row[subErrIdx] || "") : "";
                let memo = memoIdx !== -1 ? String(row[memoIdx] || "") : ""; let jenis = jenisIdx !== -1 ? String(row[jenisIdx] || "") : "";
                let comboText = `${err} ${sub} ${memo} ${jenis}`.toUpperCase();

                let category = "op"; 
                if (comboText.includes("COMPLETED") || comboText.includes("PS")) category = "comp";
                else if (comboText.includes("KENDALA PELANGGAN")) category = "kp";
                else if (comboText.includes("KENDALA TEKNIK")) category = "kt";

                // Format TSEL ID (Potong teks antara "-" pertama dan "_" pertama)
                let woVal = woIdx !== -1 ? String(row[woIdx] || "").trim() : "-";
                let tselVal = tselIdx !== -1 ? String(row[tselIdx] || "").trim() : "-";
                
                let shortTsel = tselVal;
                let matchTsel = tselVal.match(/-(.*?)(_|$)/); 
                if (matchTsel && matchTsel[1]) {
                    shortTsel = matchTsel[1];
                }
                let combinedWoTsel = `${woVal} / ${shortTsel}`;

                // Format Tim untuk Identifikasi
                let timNameFormatted = timMap[leader] || `${leader} (${member})`;

                // Save data for Modal, sertakan 'originalIndex' untuk membuka Detail Order
                let modalObj = {
                    originalIndex: idx, 
                    wo: combinedWoTsel,
                    crm: crmRaw.toUpperCase(),
                    cust: custIdx !== -1 ? row[custIdx] : "-",
                    statusText: comboText, 
                    assign: `${leader} & ${member}`,
                    teamKey: timNameFormatted
                };
                
                window.dashFilteredData.total.all.push(modalObj);
                if(window.dashFilteredData.total[type]) window.dashFilteredData.total[type].push(modalObj);
                
                window.dashFilteredData[category].all.push(modalObj);
                if(window.dashFilteredData[category][type]) window.dashFilteredData[category][type].push(modalObj);

                // Stats Increment
                stats.total.all++;
                if(stats.total[type] !== undefined) stats.total[type]++;

                stats[category].all++;
                if(stats[category][type] !== undefined) stats[category][type]++;

                // Tim Summary Increment
                if (!stats.timSummary[timNameFormatted]) stats.timSummary[timNameFormatted] = { total: 0, kp: 0, kt: 0, op: 0, comp: 0 };
                stats.timSummary[timNameFormatted].total++;
                stats.timSummary[timNameFormatted][category]++;

        function getFallbackCoords(addressStr) {
            let hash = 0;
            for (let i = 0; i < addressStr.length; i++) hash = addressStr.charCodeAt(i) + ((hash << 5) - hash);
            let latOffset = (((Math.abs(hash) % 1000) / 10000) * 0.1) - 0.05;
            let lngOffset = ((((Math.abs(hash) >> 3) % 1000) / 10000) * 0.1) - 0.05;
            let jitterLat = (Math.random() - 0.5) * 0.0015;
            let jitterLng = (Math.random() - 0.5) * 0.0015;
            return [-7.28 + latOffset + jitterLat, 112.76 + lngOffset + jitterLng];
        }

                L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', { attribution: 'OSM', subdomains: 'abcd', maxZoom: 19 }).addTo(mapInstance);
            }
            // Clear existing markers
            mapInstance.eachLayer((layer) => { if (layer instanceof L.Marker && !layer._url) mapInstance.removeLayer(layer); });

            geocodeQueue = [];
            document.getElementById('map-geo-indicator').classList.add('hidden');

            mapPoints.forEach(pt => {
                let coords = extractCoords(pt.address);
                if (coords) {
                    addHomeMarker(pt, coords[0], coords[1]);
                } else if (geocodeCache[pt.address]) {
                    if(geocodeCache[pt.address] !== 'NOT_FOUND') {
                        addHomeMarker(pt, geocodeCache[pt.address].lat, geocodeCache[pt.address].lng);
                    } else {
                        addHomeMarker(pt, pt.fallbackLat, pt.fallbackLng);
                    }
                } else {
                    geocodeQueue.push(pt);
                    addHomeMarker(pt, pt.fallbackLat, pt.fallbackLng);
                }
            });

            setTimeout(() => { mapInstance.invalidateSize(); }, 300);

            if(!isGeocoding && geocodeQueue.length > 0) {
                document.getElementById('map-geo-indicator').classList.remove('hidden');
                processGeocodeQueue();
            }
        }

        async function processGeocodeQueue() {
            if (geocodeQueue.length === 0) { 
                isGeocoding = false; 
                document.getElementById('map-geo-indicator').classList.add('hidden');
                return; 
            }
            
            isGeocoding = true;
            let pt = geocodeQueue.shift();
            
            try {
                let searchAddr = pt.address.replace(/[\n\r]+/g, ' ').trim();
                if(!searchAddr.toLowerCase().includes('surabaya')) searchAddr += ", Surabaya, Indonesia";
                else searchAddr += ", Indonesia";
                
                let response = await fetch(`https://nominatim.openstreetmap.org/search?format=json&limit=1&q=${encodeURIComponent(searchAddr)}`, { headers: { 'Accept-Language': 'id-ID' } });
                let data = await response.json();
                
                if (data && data.length > 0) {
                    geocodeCache[pt.address] = { lat: parseFloat(data[0].lat), lng: parseFloat(data[0].lon) };
                } else {
                    geocodeCache[pt.address] = 'NOT_FOUND';
                }
                
                localStorage.setItem('sympro_geocode_cache', JSON.stringify(geocodeCache));
                
            } catch(e) {
                geocodeQueue.unshift(pt); 
                await new Promise(r => setTimeout(r, 3000));
            }
            
            setTimeout(processGeocodeQueue, 1500);
        }

                    let leader = leaderIdx !== -1 ? String(row[leaderIdx] || "").trim() : ""; let member = memberIdx !== -1 ? String(row[memberIdx] || "").trim() : ""; let isAssigned = (leader !== "" && leader !== "-") && (member !== "" && member !== "-");
                    if (currentActiveMode === 'daily-order' && isAssigned) return false; if (currentActiveMode === 'order-hi' && !isAssigned) return false;

                let matchSearch = true;
                if(searchQuery !== "") { let woVal = workorderIdx !== -1 ? String(row[workorderIdx] || "").toLowerCase() : ""; let trVal = trackIdIdx !== -1 ? String(row[trackIdIdx] || "").toLowerCase() : ""; matchSearch = woVal.includes(searchQuery) || trVal.includes(searchQuery); }
                let matchCrm = true; if(selectedCrms.length > 0 && crmTypeIdx !== -1) matchCrm = selectedCrms.includes(String(row[crmTypeIdx] || "").trim());

        function processUploadExcel() {
            const fileInput = document.getElementById('excel-file'); const dataType = document.getElementById('import-type').value;
            if (fileInput.files.length === 0) { showMessage("Pilih file Excel terlebih dahulu!", "error"); return; }
            const file = fileInput.files[0]; const uploader = localStorage.getItem("sym_pro_user") || "User"; const reader = new FileReader(); document.getElementById('import-loading').classList.remove('hidden');
            reader.onload = function(e) {
                try {
                    const data = new Uint8Array(e.target.result); const workbook = XLSX.read(data, { type: 'array' }); const worksheet = workbook.Sheets[workbook.SheetNames[0]]; const rowsArray = XLSX.utils.sheet_to_json(worksheet, { header: 1, defval: "" });
                    if (rowsArray.length <= 1) { document.getElementById('import-loading').classList.add('hidden'); showMessage("File kosong.", "error"); return; }
                    if (typeof google === 'undefined') return;
                    google.script.run.withSuccessHandler(function(res) { document.getElementById('import-loading').classList.add('hidden'); if (res.success) { showMessage(res.message, 'success'); fileInput.value = ''; document.getElementById('file-chosen').innerText = "No file chosen"; loadRecentImports(); fetchGlobalUnassignedBadge(); } else showMessage(res.message, 'error'); }).importExcelData(rowsArray, dataType, uploader, file.name);
                } catch (err) { document.getElementById('import-loading').classList.add('hidden'); showMessage("Gagal membaca Excel", "error"); }
            }; reader.readAsArrayBuffer(file);
        }
