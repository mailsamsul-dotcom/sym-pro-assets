function autoResetMappingTim() {
  try {
    const ss = SpreadsheetApp.openById(getConfig('SHEET_ID'));
    const sheet = ss.getSheetByName("Mapping Tim");
    
    if (!sheet) {
      console.log("Sheet 'Mapping Tim' tidak ditemukan.");
      return;
    }
    
    const lastRow = sheet.getLastRow();
    if (lastRow > 1) {
      sheet.getRange(2, 1, lastRow - 1, sheet.getLastColumn()).clearContent();
      console.log("Sukses: Mapping Tim berhasil direset.");
    }
  } catch (e) {
    console.log("Error autoResetMappingTim: " + e.message);
  }
}

function setupTriggerResetMapping() {
  const triggers = ScriptApp.getProjectTriggers();
  for (let i = 0; i < triggers.length; i++) {
    if (triggers[i].getHandlerFunction() === "autoResetMappingTim") {
      ScriptApp.deleteTrigger(triggers[i]);
    }
  }
  ScriptApp.newTrigger("autoResetMappingTim").timeBased().everyDays(1).atHour(23).create();
}

function autoArchiveDailyOrders() {
  try {
    const ss = SpreadsheetApp.openById(getConfig('SHEET_ID'));
    const sourceSheet = ss.getSheetByName("List Daily Order");
    let sheetKendala = ss.getSheetByName("Order Kendala");
    let sheetCompleted = ss.getSheetByName("Order Completed");
    
    if (!sourceSheet) return;
    
    const data = sourceSheet.getDataRange().getValues();
    const headers = data[0].map(h => String(h).trim().toLowerCase());
    
    let jenisIdx = headers.findIndex(h => h === 'jenis_error');
    let trackIdx = headers.findIndex(h => h === 'order id tsel' || h.includes('sc order no') || h.includes('csrm'));
    let woIdx = headers.findIndex(h => h.includes('workorder'));
    
    if (jenisIdx === -1) return;

    function ensureRecordDateHeader(sheet) {
      if (!sheet) return;
      const lastCol = sheet.getLastColumn();
      if (lastCol === 0) return;
      const sHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h).trim().toUpperCase());
      let idx = sHeaders.indexOf("RECORD_DATE");
      if (idx === -1) {
        sheet.getRange(1, lastCol + 1).setValue("RECORD_DATE");
        idx = lastCol;
      }
      return idx;
    }

    function getExistingKeys(sheet) {
      const keys = {};
      if (!sheet || sheet.getLastRow() <= 1) return keys;
      const sData = sheet.getDataRange().getValues();
      const sHeaders = sData[0].map(h => String(h).trim().toLowerCase());
      let sTrackIdx = sHeaders.findIndex(h => h === 'order id tsel' || h.includes('sc order no') || h.includes('csrm'));
      let sWoIdx = sHeaders.findIndex(h => h.includes('workorder'));
      for (let i = 1; i < sData.length; i++) {
        let key = (sTrackIdx !== -1 ? String(sData[i][sTrackIdx]).trim() : "") || (sWoIdx !== -1 ? String(sData[i][sWoIdx]).trim() : "");
        if (key) keys[key] = true;
      }
      return keys;
    }

    const kendalaRecordIdx = ensureRecordDateHeader(sheetKendala);
    const completedRecordIdx = ensureRecordDateHeader(sheetCompleted);
    const existingKendala = getExistingKeys(sheetKendala);
    const existingCompleted = getExistingKeys(sheetCompleted);
    
    const rowsToKendala = [], rowsToCompleted = [], rowsToDelete = [];
    const now = new Date();
    const timestampStr = `${String(now.getDate()).padStart(2, '0')}/${String(now.getMonth() + 1).padStart(2, '0')}/${now.getFullYear()}`;

    for (let i = 1; i < data.length; i++) {
      let row = data[i];
      let rowKey = (trackIdx !== -1 ? String(row[trackIdx]).trim() : "") || (woIdx !== -1 ? String(row[woIdx]).trim() : "");
      let jenisVal = String(row[jenisIdx] || "").trim().toUpperCase();
      
      if (jenisVal.includes("COMPLETED") || jenisVal.includes("PS")) {
        if (rowKey && !existingCompleted[rowKey]) {
          let newRow = row.slice();
          while (newRow.length < completedRecordIdx) newRow.push("");
          newRow[completedRecordIdx] = timestampStr;
          rowsToCompleted.push(newRow);
          existingCompleted[rowKey] = true;
        }
        rowsToDelete.push(i + 1);
      } else if (jenisVal.includes("KENDALA")) {
        if (rowKey && !existingKendala[rowKey]) {
          let newRow = row.slice();
          while (newRow.length < kendalaRecordIdx) newRow.push("");
          newRow[kendalaRecordIdx] = timestampStr;
          rowsToKendala.push(newRow);
          existingKendala[rowKey] = true;
        }
        rowsToDelete.push(i + 1);
      }
    }

    if (rowsToKendala.length > 0) sheetKendala.getRange(sheetKendala.getLastRow() + 1, 1, rowsToKendala.length, rowsToKendala[0].length).setValues(rowsToKendala);
    if (rowsToCompleted.length > 0) sheetCompleted.getRange(sheetCompleted.getLastRow() + 1, 1, rowsToCompleted.length, rowsToCompleted[0].length).setValues(rowsToCompleted);
    if (rowsToDelete.length > 0) {
      rowsToDelete.sort((a, b) => b - a);
      rowsToDelete.forEach(r => sourceSheet.deleteRow(r));
    }
  } catch (e) {
    console.log("Error autoArchiveDailyOrders: " + e.message);
  }
}

function updateKanbanBoard(payload) {
  try {
    const ss = SpreadsheetApp.openById(getConfig('SHEET_ID'));
    const sheet = ss.getSheetByName("Order Kendala");
    if (!sheet) return { success: false, message: "Sheet tidak ditemukan." };

    const data = sheet.getDataRange().getValues();
    const headers = data[0].map(h => String(h).trim().toLowerCase());
    const trackIdx = headers.findIndex(h => h === "order id tsel" || h.includes("sc order no") || h.includes("csrm"));
    const woIdx = headers.findIndex(h => h.includes("workorder"));
    
    function ensureColumn(sheetObj, headersArr, colName) {
       let idx = headersArr.indexOf(colName.toLowerCase());
       if(idx === -1) {
          sheetObj.getRange(1, headersArr.length + 1).setValue(colName);
          headersArr.push(colName.toLowerCase());
          return headersArr.length - 1;
       }
       return idx;
    }
    
    const statusIdx = ensureColumn(sheet, headers, "KANBAN_STATUS");
    const historyIdx = ensureColumn(sheet, headers, "KANBAN_HISTORY");

    let targetRow = -1;
    for (let i = 1; i < data.length; i++) {
        let key = (trackIdx !== -1 ? String(data[i][trackIdx]).trim() : "") || (woIdx !== -1 ? String(data[i][woIdx]).trim() : "");
        if ((payload.trackId && String(data[i][trackIdx]).trim() === payload.trackId) || (payload.workorder && String(data[i][woIdx]).trim() === payload.workorder)) {
            targetRow = i + 1;
            break;
        }
    }

    if (targetRow !== -1) {
        let oldHistory = String(data[targetRow-1][historyIdx] || "");
        let newLog = "[" + payload.timestamp + "] " + payload.user + " memindahkan status ke " + payload.newStatus;
        if (payload.note && payload.note.trim() !== "") newLog += "\n📝 Update: " + payload.note.trim();
        let updatedHistory = oldHistory ? oldHistory + "\n\n" + newLog : newLog;
        
        sheet.getRange(targetRow, statusIdx + 1).setValue(payload.newStatus);
        sheet.getRange(targetRow, historyIdx + 1).setValue(updatedHistory);
        return { success: true, message: "Berhasil update progres" };
    }
    return { success: false, message: "Data tidak ditemukan." };
  } catch(e) {
    return { success: false, message: e.message };
  }
}

function getAllConfigs() {

}

function saveAllConfigs(configs) {
  try {
    const ss = SpreadsheetApp.openById(getConfig('SHEET_ID'));
    const sheet = ss.getSheetByName("Config");
    sheet.getRange(1, 1, sheet.getLastRow(), 2).clearContent();
    sheet.getRange(1, 1, configs.length, 2).setValues(configs);
    return { success: true, message: "Konfigurasi berhasil disimpan!" };
  } catch(e) {
    return { success: false, message: e.message };
  }
}
